[90. 子集 II](https://leetcode.cn/problems/subsets-ii/)

给你一个整数数组 `nums` ，其中可能包含重复元素，请你返回该数组所有可能的 子集（幂集）。

解集 **不能** 包含重复的子集。返回的解集中，子集可以按 **任意顺序** 排列。

 

**示例 1：**

```
输入：nums = [1,2,2]
输出：[[],[1],[1,2],[1,2,2],[2],[2,2]]
```

**示例 2：**

```
输入：nums = [0]
输出：[[],[0]]
```

 

**提示：**

- `1 <= nums.length <= 10`
- `-10 <= nums[i] <= 10`



## 解法：回溯

```c
/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
int* path;
int path_len;

int** ret;
int ret_len;

bool* used;

int cmp(const void* pa, const void* pb)
{
    return *(int*)pa - *(int*)pb;
}

void backtracking(int* nums, int numsSize, int start, int* returnColumnSizes)
{
    ret[ret_len] = (int*)malloc(path_len * sizeof(int));
    for (int i = 0; i < path_len; ++i) ret[ret_len][i] = path[i];
    returnColumnSizes[ret_len] = path_len;
    ++ret_len;

    if (start == numsSize) return;

    for (int i = start; i < numsSize; ++i)
    {
        // 树层去重，树枝不去重
        if (i > start && nums[i - 1] == nums[i] && used[i - 1] == false)
            continue;
        
        path[path_len++] = nums[i];
        used[i] = true;

        backtracking(nums, numsSize, i + 1, returnColumnSizes);

        --path_len;
        used[i] = false;
    }
}

int** subsetsWithDup(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    qsort(nums, numsSize, sizeof(int), cmp);
    
    path = (int*)malloc(numsSize * sizeof(int));
    ret = (int**)malloc((1 << numsSize) * sizeof(int*));
    *returnColumnSizes = (int*)malloc((1 << numsSize) * sizeof(int));
    path_len = ret_len = 0;

    used = (bool*)malloc(numsSize * sizeof(bool));
    for (int i = 0; i < numsSize; ++i) used[i] = false;

    backtracking(nums, numsSize, 0, *returnColumnSizes);

    *returnSize = ret_len;
    free(path);
    return ret;
}
```
