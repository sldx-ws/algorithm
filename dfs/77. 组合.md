[77. 组合](https://leetcode.cn/problems/combinations/)	

给定两个整数 `n` 和 `k`，返回范围 `[1, n]` 中所有可能的 `k` 个数的组合。

你可以按 **任何顺序** 返回答案。

 

**示例 1：**

```
输入：n = 4, k = 2
输出：
[
  [2,4],
  [3,4],
  [2,3],
  [1,2],
  [1,3],
  [1,4],
]
```

**示例 2：**

```
输入：n = 1, k = 1
输出：[[1]]
```

 

**提示：**

- `1 <= n <= 20`
- `1 <= k <= n`



## 解法：回溯

```c
/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
int** ret;
int ret_len;

int* path;
int path_len;

void backtracking(int n, int k, int cur)
{
    if (path_len == k)
    {
        ret[ret_len] = (int*)malloc(k * sizeof(int));
        for (int i = 0; i < k; ++i) 
        {
            ret[ret_len][i] = path[i];
        }
        ++ret_len;

        return;
    }

    // for (int i = cur; i <= n; ++i)
    for (int i = cur; n - i + 1 >= k - path_len; ++i) // 剪枝:剩余元素个数>=所需元素个数
    {
        path[path_len++] = i;
        backtracking(n, k, i + 1);

        // 回溯
        --path_len;
    }
}

int** combine(int n, int k, int* returnSize, int** returnColumnSizes) {
    path = (int*)malloc(k * sizeof(int));
    ret = (int**)malloc(200001 * sizeof(int*));
    path_len = ret_len = 0;

    backtracking(n, k, 1);

    *returnSize = ret_len;
    *returnColumnSizes = (int*)malloc(ret_len * sizeof(int));
    for (int i = 0; i < *returnSize; ++i) (*returnColumnSizes)[i] = k;

    free(path);
    return ret;
}
```
