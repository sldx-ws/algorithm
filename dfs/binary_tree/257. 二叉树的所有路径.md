[257. 二叉树的所有路径](https://leetcode.cn/problems/binary-tree-paths/)

给你一个二叉树的根节点 `root` ，按 **任意顺序** ，返回所有从根节点到叶子节点的路径。

**叶子节点** 是指没有子节点的节点。

**示例 1：**

![img](https://assets.leetcode.com/uploads/2021/03/12/paths-tree.jpg)

```
输入：root = [1,2,3,null,5]
输出：["1->2->5","1->3"]
```

**示例 2：**

```
输入：root = [1]
输出：["1"]
```

 

**提示：**

- 树中节点的数目在范围 `[1, 100]` 内
- `-100 <= Node.val <= 100`



## 解法：dfs

```c
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     struct TreeNode *left;
 *     struct TreeNode *right;
 * };
 */
/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
char* path;
int path_len;

char** ret;
int ret_len;

void dfs(struct TreeNode* root)
{
    if (!root) return;
    
    int old_len = path_len;  // 记录当前长度，用于回溯
    path_len += sprintf(path + path_len, "%d", root->val);  // 注意sprintf返回值
    
    if (!root->left && !root->right)
    {
        // path[pathSize++] = '\0';  // sprintf自动在末尾加'\0'
        ret[ret_len] = (char*)malloc((path_len + 1) * sizeof(char));
        strcpy(ret[ret_len], path);
        ++ret_len;
        
        path_len = old_len;  // 回溯

        return;
    }

    path_len += sprintf(path + path_len, "->");

    dfs(root->left);
    dfs(root->right);

    path_len = old_len;  // 回溯
}
 
char** binaryTreePaths(struct TreeNode* root, int* returnSize) {
    path = (char*)calloc(101, sizeof(char));
    ret = (char**)malloc(101 * sizeof(char*));
    path_len = ret_len = 0;

    dfs(root);

    free(path);
    *returnSize = ret_len;
    return ret;
}
```

### 补充

```c
int sprintf(char *str, const char *format, ...);
// sprintf会自动在末尾加'\0'
// 返回值：成功写入的字符数量（不包含末尾自动添加的'\0'）
```
